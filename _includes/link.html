{%- comment -%} A link that opens external URLs (anything starting with "http") in a new tab. Usage: {% include link.html url="..." label="..." %} {%- endcomment -%} {%- assign scheme = include.url | slice: 0, 4 -%} {%- if scheme == "http" -%} {{ include.label }} (opens in a new tab) {%- else -%} {{ include.label }} {%- endif -%}
